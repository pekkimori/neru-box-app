**Goal:** Convert the current Social tab into a playful Gacha tab with coin-based creature pulls, a Pokédex subpage, and a simulated friend exchange subpage.

- [ ] **Step 3: Compute ownership and filters**

Inside `PokedexScreen`:

```tsx
const { gachaResults } = useNeru();
const [filter, setFilter] = useState<Filter>('all');
const [selected, setSelected] = useState<Creature | null>(null);

const ownedCounts = useMemo(() => {
  return gachaResults.reduce<Record<string, number>>((acc, result) => {

    return acc;
  }, {});
```

```tsx
          </>
```

- [ ] **Step 3: Compute duplicate inventory**

Inside `ExchangeScreen`:

```tsx
const { gachaResults } = useNeru();
const [offers, setOffers] = useState(INITIAL_OFFERS);

const ownedCounts = useMemo(() => {
  return gachaResults.reduce<Record<string, number>>((acc, result) => {
    acc[result.name] = (acc[result.name] || 0) + 1;
    return acc;
  }, {});
}, [gachaResults]);

const tradeableCreatures = Object.entries(ownedCounts)
  .filter(([, count]) => count > 1)
  .map(([name, count]) => {
    const creature = CREATURES.find(item => item.name === name);
    return { name, count, emoji: creature?.emoji ?? '?', rarity: creature?.rarity ?? 'common' as Rarity };
  });

const dismissOffer = (offerId: string) => {
  setOffers(current => current.filter(offer => offer.id !== offerId));
};
```
